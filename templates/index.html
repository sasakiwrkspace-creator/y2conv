===================================== # バイト → MB # ===================================== def bytes_to_mb(value: int) -> float: return value / ( 1024 * 1024 ) # ===================================== # meminfoは通常kB if len(parts) >= 2: unit = parts[1].lower() if unit == "kb": number *= 1024 values[key] = number return values except Exception: return None # ===================================== # システムメモリログ # ===================================== # FFmpeg RAM監視 # ===================================== class MemoryMonitor: def __init__( self, pid: int ): self.pid = pid self.stop_event = ( threading.Event() ) self.thread = None self.max_rss = 0 self.last_rss = 0 self.warning_logged = False self.critical_logged = False def start(self): self.thread = threading.Thread( target=self._run, name="ffmpeg-memory-monitor", daemon=True, ) self.thread.start() log( "FFmpeg RAM監視スレッド開始" ) def stop(self): self.stop_event.set() if self.thread is not None: self.thread.join( timeout=3 ) log( "FFmpeg RAM監視スレッド終了" ) def _run(self): while not self.stop_event.is_set(): rss = get_process_rss_bytes( self.pid ) if rss is not None: self.last_rss = rss if rss > self.max_rss: self.max_rss = rss rss_mb = bytes_to_mb( rss ) max_mb = bytes_to_mb( self.max_rss ) limit_mb = bytes_to_mb( MEMORY_LIMIT_BYTES ) log( f"FFmpeg RAM: " f"{rss_mb:.1f} MB " f"/ max {max_mb:.1f} MB " f"/ limit {limit_mb:.1f} MB" ) # ----------------------------- # 90% # ----------------------------- if ( rss >= MEMORY_WARNING_BYTES and not self.warning_logged ): self.warning_logged = True log( "WARNING: " "FFmpeg RAMが512MB制限の90%を超えました" ) # ----------------------------- # 95% # ----------------------------- if ( rss >= MEMORY_CRITICAL_BYTES and not self.critical_logged ): self.critical_logged = True log( "CRITICAL: " "FFmpeg RAMが512MB制限の95%を超えました" ) else: # プロセス終了後は # /proc/PID/statusが消えるので # 静かに終了する。 if not Path( f"/proc/{self.pid}" ).exists(): break self.stop_event.wait( MEMORY_LOG_INTERVAL ) # ===================================== # モジュールロード確認 # ===================================== separator() log( "subtitle_test_ffmpeg.py " "MODULE LOAD COMPLETE" ) log( f"Current working directory: " f"{Path.cwd()}" ) log( f"BASE_DIR: {BASE_DIR}" ) log( f"DOWNLOAD_DIR: {DOWNLOAD_DIR}" ) log( f"TEST_MP4: {TEST_MP4}" ) log( f"TEST_SRT: {TEST_SRT}" ) log( f"OUTPUT_MP4: {OUTPUT_MP4}" ) separator() #
