# ===================================== # subtitle_test_ffmpeg.py # # 最小FFmpeg起動テスト # # 入力: # /app/downloads/test.mp4 # # 出力: # /app/downloads/test_copy.mp4 # # 今回の目的: # # 1. このPythonファイルが実際に実行されているか確認 # 2. run_ffmpeg() に到達しているか確認 # 3. subprocess.Popen() が実行されているか確認 # 4. 512MB MEMORY_LIMIT_BYTES = ( 512 * 1024 * 1024 ) # 90% MEMORY_WARNING_BYTES = int( MEMORY_LIMIT_BYTES * 0.90 ) # 95% MEMORY_CRITICAL_BYTES = int( MEMORY_LIMIT_BYTES * 0.95 ) # 1秒ごとに表示 MEMORY_LOG_INTERVAL = 1.0 # ===================================== # ログ # ===================================== def log(message: str = ""): print( f"[FFMPEG TEST] {message}", flush=True, ) def separator(): print( "[FFMPEG TEST] ==========================================", flush=True, ) # ===================================== # Bytes → MB # ===================================== def bytes_to_mb( value: int, ) -> float: return ( value / (1024 * 1024) ) # ===================================== # Pythonプロセス情報 # ===================================== def print_python_info(): separator() log( "PYTHON / ENVIRONMENT INFO" ) log( f"Python PID: {os.getpid()}" ) log( f"Python executable: " f"{os.sys.executable}" ) log( f"Current working directory: " f"{Path.cwd()}" ) log( f"BASE_DIR: " f"{BASE_DIR}" ) log( f"DOWNLOAD_DIR: " f"{DOWNLOAD_DIR}" ) log( f"INPUT_MP4: " f"{INPUT_MP4}" ) log( f"OUTPUT_MP4: " f"{OUTPUT_MP4}" ) log( f"MEMORY LIMIT: " f"{bytes_to_mb(MEMORY_LIMIT_BYTES):.1f} MB" ) log( f"MEMORY WARNING: " f"{bytes_to_mb(MEMORY_WARNING_BYTES):.1f} MB" ) log( f"MEMORY CRITICAL: " f"{bytes_to_mb(MEMORY_CRITICAL_BYTES):.1f} MB" ) separator() # ===================================== # FFmpeg RSS取得 # ===================================== def get_process_rss_bytes( pid: int, ) -> int | None: status_path = Path( f"/proc/{pid}/status" ) try: if not status_path.exists(): return None text = status_path.read_text( encoding="utf-8", errors="replace", ) for line in text.splitlines(): if not line.startswith( "VmRSS:" ): continue parts = line.split() if len(parts) < 2: return None # LinuxのVmRSSは通常kB kb = int( parts[1] ) return ( kb * 1024 ) except Exception as e: log( f"[MEMORY] RSS取得エラー: " f"{type(e).__name__}: {e}" ) return None return None # ===================================== # FFmpegメモリ監視 # ===================================== class FFmpegMemoryMonitor: def __init__( self, pid: int, ): self.pid = pid self.stop_event = ( threading.Event() ) self.thread = None self.max_rss = 0 self.last_rss = 0 self.warning_logged = False self.critical_logged = False # ================================= # 監視ループ # ================================= def _monitor_loop(self): log( "[MEMORY] 監視スレッド開始" ) log( f"[MEMORY] 監視対象PID: " f"{self.pid}" ) while not self.stop_event.is_set(): rss = ( get_process_rss_bytes( self.pid ) ) # ============================= # RSS取得成功 # ============================= if rss is not None: self.last_rss = rss if rss > self.max_rss: self.max_rss = rss rss_mb = ( bytes_to_mb( rss ) ) max_mb = ( bytes_to_mb( self.max_rss ) ) limit_mb = ( bytes_to_mb( MEMORY_LIMIT_BYTES ) ) log( "[MEMORY] " f"FFmpeg RSS: " f"{rss_mb:.1f} MB " f"/ MAX: " f"{max_mb:.1f} MB " f"/ LIMIT: " f"{limit_mb:.1f} MB" ) # ========================= # 90% # ========================= if ( rss >= MEMORY_WARNING_BYTES and not self.warning_logged ): self.warning_logged = True log( "[MEMORY] " "WARNING: " "FFmpeg RSSが512MBの90%を超えました" ) # ========================= # 95% # ========================= if ( rss >= MEMORY_CRITICAL_BYTES and not self.critical_logged ): self.critical_logged = True log( "[MEMORY] " "CRITICAL: " "FFmpeg RSSが512MBの95%を超えました" ) # ========================= # 512MB超過 # ========================= if ( rss >= MEMORY_LIMIT_BYTES ): log( "[MEMORY] " "!!! CRITICAL: " "FFmpeg RSSが512MBを超えています !!!" ) # ============================= # RSS取得不可 # ============================= else: proc_path = Path( f"/proc/{self.pid}" ) if not proc_path.exists(): log( "[MEMORY] " "FFmpegプロセスが終了したため監視終了" ) break log( "[MEMORY] " "RSS取得不可" ) self.stop_event.wait( MEMORY_LOG_INTERVAL ) log( "[MEMORY] 監視スレッド終了" ) # ===================================== # 入力ファイル確認 # ===================================== def check_input(): separator() log( "STEP 1: 入力ファイル確認" ) log( f"INPUT: {INPUT_MP4}" ) log( f"exists: {INPUT_MP4.exists()}" ) if not INPUT_MP4.exists(): raise FileNotFoundError( f"入力ファイルがありません: " f"{INPUT_MP4}" ) if not INPUT_MP4.is_file(): raise RuntimeError( f"入力パスがファイルではありません: " f"{INPUT_MP4}" ) size = ( INPUT_MP4.stat().st_size ) log( f"INPUT SIZE: " f"{size} bytes" ) log( f"INPUT SIZE: " f"{bytes_to_mb(size):.3f} MB" ) if size <= 0: raise RuntimeError( "入力ファイルサイズが0です" ) log( "入力ファイル確認OK" ) # ===================================== # FFmpeg確認 # ===================================== def check_ffmpeg(): separator() log( "STEP 2: FFmpeg確認" ) log( "shutil.which('ffmpeg') 実行" ) ffmpeg = shutil.which( "ffmpeg" ) log( f"FFmpeg PATH: {ffmpeg}" ) if not ffmpeg: raise RuntimeError( "FFmpegが見つかりません" ) log( "FFmpeg version確認開始" ) result = subprocess.run( [ ffmpeg, "-version", ], capture_output=True, text=True, timeout=10, ) log( f"FFmpeg version returncode: " f"{result.returncode}" ) if result.stdout: first_line = ( result.stdout .splitlines()[0] ) log( f"FFmpeg version: " f"{first_line}" ) if result.returncode != 0: raise RuntimeError( "FFmpeg version確認失敗" ) log( "FFmpeg確認OK" ) return ffmpeg # ===================================== # FFmpegコマンド作成 # ===================================== def create_command( ffmpeg: str, ): separator() log( "STEP 3: FFmpegコマンド作成" ) # --------------------------------- # 最小構成 # # 再エンコードなし # 字幕なし # filterなし # --------------------------------- command = [ ffmpeg, "-y", "-nostdin", "-hide_banner", "-loglevel", "info", "-i", str(INPUT_MP4), "-c", "copy", str(OUTPUT_MP4), ] log( "FFmpegコマンド:" ) log( " ".join( str(x) for x in command ) ) log( "FFmpegコマンド作成完了" ) return command # ===================================== # run_ffmpeg # ===================================== def run_ffmpeg( ffmpeg: str, command: list[str], ): separator() log( "★★★★★ run_ffmpeg() ENTER ★★★★★" ) log( f"FFmpeg executable: " f"{ffmpeg}" ) log( f"command length: " f"{len(command)}" ) log( "★★★★★ run_ffmpeg() ENTER END ★★★★★" ) start_time = ( time.monotonic() ) process = None memory_monitor = None output_lines = [] # ================================= # Popen直前 # ================================= separator() log( "STEP 4: subprocess.Popen() 直前" ) log( "★ これからFFmpegプロセスを起動します ★" ) log( "COMMAND:" ) log( " ".join( str(x) for x in command ) ) try: # ============================= # Popen # ============================= log( "★★★★★ subprocess.Popen() CALL ★★★★★" ) process = subprocess.Popen( command, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, bufsize=1, ) # ============================= # Popen完了 # ============================= log( "★★★★★ subprocess.Popen() RETURN ★★★★★" ) log( "FFmpegプロセス起動成功" ) log( f"FFmpeg PID: " f"{process.pid}" ) # ================================= # PID存在確認 # ================================= proc_path = Path( f"/proc/{process.pid}" ) log( f"/proc/{process.pid} exists: " f"{proc_path.exists()}" ) # ================================= # 初期RSS # ================================= log( "STEP 5: FFmpeg初期RSS確認" ) initial_rss = ( get_process_rss_bytes( process.pid ) ) if initial_rss is not None: log( "[MEMORY] " "FFmpeg開始時RSS: " f"{bytes_to_mb(initial_rss):.1f} MB" ) else: log( "[MEMORY] " "FFmpeg開始時RSS: 取得不可" ) # ================================= # MemoryMonitor作成 # ================================= separator() log( "STEP 6: MemoryMonitor作成" ) log( "★★★★★ MemoryMonitor作成前 ★★★★★" ) memory_monitor = ( FFmpegMemoryMonitor( process.pid ) ) log( "★★★★★ MemoryMonitor作成完了 ★★★★★" ) # ================================= # MemoryMonitor開始 # ================================= log( "★★★★★ MemoryMonitor.start() 前 ★★★★★" ) memory_monitor.start() log( "★★★★★ MemoryMonitor.start() 完了 ★★★★★" ) # ================================= # stdout # ================================= separator() log( "STEP 7: FFmpegログ読み取り開始" ) log( "★ FFmpegは現在実行中です ★" ) assert process.stdout is not None # ================================= # ログ読み取り # ================================= while True: # ----------------------------- # timeout確認 # ----------------------------- elapsed = ( time.monotonic() - start_time ) if ( elapsed > FFMPEG_TIMEOUT ): log( "!!! FFmpeg TIMEOUT !!!" ) log( f"elapsed: " f"{elapsed:.3f} sec" ) log( f"PID: " f"{process.pid}" ) try: process.kill() log( "process.kill() 実行" ) except Exception as e: log( f"process.kill() FAILED: " f"{e}" ) try: process.wait( timeout=10 ) except Exception as e: log( f"process.wait() FAILED: " f"{e}" ) raise TimeoutError( f"FFmpegが" f"{FFMPEG_TIMEOUT}秒以内に" f"終了しませんでした" ) # ----------------------------- # stdout読み取り # ----------------------------- line = ( process.stdout.readline() ) if line: line = line.rstrip() output_lines.append( line ) print( f"[FFMPEG] {line}", flush=True ) continue # ----------------------------- # stdout EOF # ----------------------------- returncode = ( process.poll() ) if returncode is not None: log( "FFmpegプロセス終了を検出" ) break # ----------------------------- # 少し待つ # ----------------------------- time.sleep( 0.05 ) # ================================= # wait # ================================= log( "STEP 8: process.wait()" ) returncode = ( process.wait() ) log( f"process.wait() returncode: " f"{returncode}" ) # ================================= # MemoryMonitor停止 # ================================= if memory_monitor: log( "★★★★★ MemoryMonitor.stop() 前 ★★★★★" ) memory_monitor.stop() log( "★★★★★ MemoryMonitor.stop() 完了 ★★★★★" ) # ================================= # 経過時間 # ================================= elapsed = ( time.monotonic() - start_time ) # ================================= # 最大RSS # ================================= separator() log( "STEP 9: FFmpeg終了情報" ) log( f"FFmpeg returncode: " f"{returncode}" ) log( f"FFmpeg elapsed: " f"{elapsed:.3f} sec" ) if memory_monitor: max_rss = ( memory_monitor.max_rss ) log( "------------------------------------------" ) log( "FFmpeg最大メモリ使用量" ) log( f"MAX RSS: " f"{bytes_to_mb(max_rss):.1f} MB" ) remaining = ( MEMORY_LIMIT_BYTES - max_rss ) log( f"512MBまで残り: " f"{bytes_to_mb(remaining):.1f} MB" ) if max_rss >= MEMORY_LIMIT_BYTES: log( "!!! FFmpeg FAILED !!!" ) log( "最後のFFmpegログ:" ) for line in output_lines[-30:]: log( f" {line}" ) raise RuntimeError( "FFmpeg failed: " f"returncode={returncode}" ) # ================================= # 出力確認 # ================================= separator() log( "STEP 10: 出力ファイル確認" ) log( f"OUTPUT: " f"{OUTPUT_MP4}" ) if not OUTPUT_MP4.exists(): raise RuntimeError( "FFmpeg returncode=0ですが" "出力ファイルがありません" ) output_size = ( OUTPUT_MP4.stat().st_size ) log( f"OUTPUT SIZE: " f"{output_size} bytes" ) log( f"OUTPUT SIZE: " f"{bytes_to_mb(output_size):.3f} MB" ) if output_size <= 0: raise RuntimeError( "出力ファイルサイズが0です" ) #
