# ===================================== # ffmpeg_test.py # # 最小FFmpeg動作テスト # # 目的: # ブラウザのボタンを押す # ↓ # PythonがFFmpegを実行 # ↓ # FFmpeg終了 # ↓ # ブラウザに「FFmpeg終了」を表示 # # 字幕処理: # 一切しない # # SRT: # 一切使用しない # # libass: # 一切使用しない # # フォント: # 一切使用しない # # メモリ監視: # 一切しない # ===================================== from __future__ import annotations import os import shutil import subprocess import time from flask import Flask, jsonify, render_template_string # ===================================== # Flask # ===================================== app = Flask(__name__) # ===================================== # 設定 # ===================================== BASE_DIR = os.environ.get( "BASE_DIR", "/app" ) DOWNLOAD_DIR = os.environ.get( "DOWNLOAD_DIR", os.path.join( BASE_DIR, "downloads" ) ) INPUT_FILE = os.path.join( DOWNLOAD_DIR, "test.mp4" ) OUTPUT_FILE = os.path.join( DOWNLOAD_DIR, "ffmpeg_test_output.mp4" ) FFMPEG_TIMEOUT = 120 # ===================================== # ログ # ===================================== def log(message: str): print( f"[FFMPEG TEST] {message}", flush=True ) # ===================================== # ブラウザ画面 # ===================================== HTML = """ FFmpeg Test

FFmpeg単体テスト

字幕処理は一切行いません。

まだ実行していません。
""" # ===================================== # トップページ # ===================================== @app.route("/") def index(): log( "ブラウザからトップページへのアクセス" ) return render_template_string( HTML ) # ===================================== # FFmpeg実行 # ===================================== @app.route( "/run-ffmpeg", methods=["POST"] ) def run_ffmpeg(): log("") log( "==========================================" ) log( "ブラウザからFFmpeg実行要求を受信" ) log( "==========================================" ) start_time = time.monotonic() # ================================= # FFmpeg確認 # ================================= log( "STEP 1: FFmpeg検索" ) ffmpeg = shutil.which( "ffmpeg" ) log( f"ffmpeg: {ffmpeg}" ) if not ffmpeg: log( "ERROR: FFmpegが見つかりません" ) return jsonify( { "success": False, "message": "FFmpegが見つかりません" } ), 500 # ================================= # 入力確認 # ================================= log( "STEP 2: 入力ファイル確認" ) log( f"INPUT_FILE: {INPUT_FILE}" ) if not os.path.exists( INPUT_FILE ): log( "ERROR: test.mp4がありません" ) return jsonify( { "success": False, "message": f"入力ファイルがありません: " f"{INPUT_FILE}" } ), 400 input_size = os.path.getsize( INPUT_FILE ) log( f"入力ファイルサイズ: " f"{input_size} bytes" ) # ================================= # 出力削除 # ================================= log( "STEP 3: 既存出力削除" ) if os.path.exists( OUTPUT_FILE ): log( f"既存ファイル削除: " f"{OUTPUT_FILE}" ) os.remove( OUTPUT_FILE ) # ================================= # FFmpegコマンド # # 字幕なし # フィルターなし # # test.mp4をそのまま # H.264 + AACに変換するだけ。 # ================================= command = [ ffmpeg, "-y", "-nostdin", "-hide_banner", "-loglevel", "info", "-i", INPUT_FILE, "-c:v", "libx264", "-preset", "ultrafast", "-crf", "30", "-c:a", "aac", "-b:a", "128k", "-movflags", "+faststart", OUTPUT_FILE, ] log( "STEP 4: FFmpegコマンド" ) log( " ".join(command) ) # ================================= # FFmpeg開始 # ================================= log( "STEP 5: FFmpeg開始" ) log( "ここからFFmpeg subprocess.Popen()" ) try: process = subprocess.Popen( command, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, bufsize=1, ) except Exception as e: log( "FFmpeg Popen() FAILED" ) log( f"ERROR TYPE: " f"{type(e).__name__}" ) log( f"ERROR: {e}" ) return jsonify( { "success": False, "message": f"FFmpeg起動失敗: {e}" } ), 500 # ================================= # PID # ================================= log( f"FFmpeg PID: {process.pid}" ) log( "FFmpegプロセスは起動しました" ) # ================================= # FFmpegログ読み取り # ================================= ffmpeg_logs = [] try: if process.stdout is not None: for line in process.stdout: line = line.rstrip() ffmpeg_logs.append( line ) print( f"[FFMPEG] {line}", flush=True ) # ログが増えすぎないように # 最後の100行だけ保持 if len(ffmpeg_logs) > 100: ffmpeg_logs.pop(0) # ------------------------- # timeout # ------------------------- elapsed = ( time.monotonic() - start_time ) if ( elapsed > FFMPEG_TIMEOUT ): log( "FFmpeg TIMEOUT" ) log( f"経過時間: " f"{elapsed:.1f} 秒" ) try: process.kill() log( "process.kill() 実行" ) except Exception as e: log( f"kill失敗: {e}" ) process.wait( timeout=10 ) return jsonify( { "success": False, "message": "FFmpegが" f"{FFMPEG_TIMEOUT}秒" "以内に終了しませんでした", "pid": process.pid, "logs": ffmpeg_logs, } ), 500 except Exception as e: log( "FFmpegログ読み取り中に例外" ) log( f"ERROR TYPE: " f"{type(e).__name__}" ) log( f"ERROR: {e}" ) try: process.kill() except Exception: pass try: process.wait( timeout=10 ) except Exception: pass return jsonify( { "success": False, "message": f"FFmpegログ読み取り失敗: {e}", "pid": process.pid, "logs": ffmpeg_logs, } ), 500 # ================================= # FFmpeg終了待ち # ================================= log( "STEP 6: FFmpeg終了待ち" ) try: returncode = process.wait( timeout=10 ) except subprocess.TimeoutExpired: log( "FFmpeg終了待ちTIMEOUT" ) try: process.kill() except Exception: pass return jsonify( { "success": False, "message": "FFmpegプロセスが終了しません", "pid": process.pid, } ), 500 # ================================= # 終了 # ================================= elapsed = ( time.monotonic() - start_time ) log( "STEP 7: FFmpeg終了" ) log( f"returncode: {returncode}" ) log( f"elapsed: {elapsed:.3f} 秒" ) # ================================= # returncode確認 # ================================= if returncode != 0: log( "FFmpeg FAILED" ) log( "最後のFFmpegログ:" ) for line in ffmpeg_logs[-30:]: log( f"[FFMPEG-LAST] {line}" ) return jsonify( { "success": False, "message": f"FFmpeg失敗 " f"(returncode={returncode})", "returncode": returncode, "elapsed": round( elapsed, 3 ), "logs": ffmpeg_logs, } ), 500 # ================================= # 出力確認 # ================================= log( "STEP 8: FFmpeg出力確認" ) if not os.path.exists( OUTPUT_FILE ): log( "ERROR: FFmpegはreturncode=0ですが" "出力ファイルがありません" ) return jsonify( { "success": False, "message": "FFmpegは終了しましたが" "出力ファイルがありません", "returncode": returncode, "elapsed": round( elapsed, 3 ), } ), 500 output_size = os.path.getsize( OUTPUT_FILE ) log( f"出力ファイル: " f"{OUTPUT_FILE}" ) log( f"出力サイズ: " f"{output_size} bytes" ) # ================================= # ★ ここが今回の確認ポイント # ================================= log( "==========================================" ) log( "FFmpeg終了" ) log( "==========================================" ) return jsonify( { "success": True, "message": "FFmpeg終了", "returncode": returncode, "elapsed": round( elapsed, 3 ), "output": OUTPUT_FILE, "output_size": output_size, "logs": ffmpeg_logs[-30:], } ) # ===================================== # 起動 # ===================================== if __name__ == "__main__": log( "==========================================" ) log( "ffmpeg_test.py START" ) log( f"BASE_DIR: {BASE_DIR}" ) log( f"DOWNLOAD_DIR: {DOWNLOAD_DIR}" ) log( f"INPUT_FILE: {INPUT_FILE}" ) log( f"OUTPUT_FILE: {OUTPUT_FILE}" ) log( f"FFMPEG_TIMEOUT: " f"{FFMPEG_TIMEOUT}" ) log( "==========================================" ) app.run( host="0.0.0.0", port=int( os.environ.get( "PORT", "10000" ) ), debug=False, threaded=True, )
