from flask import Flask, jsonify, render_template_string import subprocess import os app = Flask(__name__) # ========================================== # ブラウザ画面 # ========================================== HTML = """ FFmpeg Test

FFmpegテスト

待機中

""" # ========================================== # 起動確認 # ========================================== print( "==========================================", flush=True ) print( "[SUBTITLE TEST] subtitle_test_ffmpeg.py START", flush=True ) print( f"[SUBTITLE TEST] Python PID: {os.getpid()}", flush=True ) print( f"[SUBTITLE TEST] __file__: {__file__}", flush=True ) print( "==========================================", flush=True ) # ========================================== # トップページ # ========================================== @app.route("/") def index(): print( "[SUBTITLE TEST] ブラウザからページ要求", flush=True ) return render_template_string( HTML ) # ========================================== # FFmpeg実行 # ========================================== @app.route( "/run-ffmpeg", methods=["POST"] ) def run_ffmpeg(): print( "==========================================", flush=True ) print( "[SUBTITLE TEST] /run-ffmpeg 開始", flush=True ) print( "[SUBTITLE TEST] FFmpegを起動します", flush=True ) try: # -------------------------------------- # FFmpegを実行 # # 動画処理は一切しない。 # バージョン確認だけ。 # -------------------------------------- result = subprocess.run( [ "/usr/bin/ffmpeg", "-version" ], stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True, timeout=30, ) print( "[SUBTITLE TEST] FFmpegプロセス終了", flush=True ) print( f"[SUBTITLE TEST] returncode: " f"{result.returncode}", flush=True ) # -------------------------------------- # FFmpegバージョン # -------------------------------------- if result.stdout: lines = ( result.stdout .splitlines() ) if lines: print( f"[FFMPEG] {lines[0]}", flush=True ) # -------------------------------------- # 成功 # -------------------------------------- if result.returncode == 0: print( "[SUBTITLE TEST] " "FFmpeg終了 → ブラウザへ返します", flush=True ) print( "==========================================", flush=True ) return jsonify( { "success": True, "message": "FFmpeg終了" } ) # -------------------------------------- # FFmpegエラー # -------------------------------------- error_message = ( result.stderr.strip() if result.stderr else f"returncode={result.returncode}" ) print( "[SUBTITLE TEST] " f"FFmpeg ERROR: {error_message}", flush=True ) print( "==========================================", flush=True ) return jsonify( { "success": False, "error": error_message } ), 500 # ------------------------------------------ # timeout # ------------------------------------------ except subprocess.TimeoutExpired: print( "[SUBTITLE TEST] " "FFmpeg TIMEOUT", flush=True ) return jsonify( { "success": False, "error": "FFmpeg TIMEOUT" } ), 500 # ------------------------------------------ # その他のエラー # ------------------------------------------ except Exception as e: print( "[SUBTITLE TEST] " "FFmpeg実行例外", flush=True ) print( f"[SUBTITLE TEST] " f"ERROR TYPE: {type(e).__name__}", flush=True ) print( f"[SUBTITLE TEST] " f"ERROR: {e}", flush=True ) return jsonify( { "success": False, "error": str(e) } ), 500 # ========================================== # 直接実行 # ========================================== if __name__ == "__main__": print( "[SUBTITLE TEST] " "直接実行モード", flush=True ) app.run( host="0.0.0.0", port=10000, debug=False, )
